Словарь записей по ID · пример

def add_record(book, record):
    key = record["id"]
    if key in book:
        return False
    book[key] = record
    return True

book = {}
print(add_record(book, {"id": "S01", "name": "Анна"}))
print(add_record(book, {"id": "S01", "name": "Борис"}))
print(book["S01"]["name"])

Поиск и фильтрация · пример

def select_group(students, group):
    result = []
    for student in students:
        if student["group"] == group:
            result.append(student)
    return result

rows = [{"name": "Анна", "group": "A"},
        {"name": "Борис", "group": "B"}]
print(select_group(rows, "A"))

Группировка и подсчёт · пример

students = [{"name": "Анна", "group": "A"},
            {"name": "Борис", "group": "B"},
            {"name": "Вера", "group": "A"}]
counts = {}
for student in students:
    group = student["group"]
    counts[group] = counts.get(group, 0) + 1
print(counts)

Упорядоченный отчёт · пример

rows = [{"id": "2", "name": "анна"},
        {"id": "1", "name": "Анна"},
        {"id": "3", "name": "Борис"}]
ordered = sorted(rows, key=lambda row: (row["name"].casefold(), row["id"]))
print([row["id"] for row in ordered])
